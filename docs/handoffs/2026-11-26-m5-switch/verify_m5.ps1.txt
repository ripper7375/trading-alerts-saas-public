# Check of the M5 switch against the production database (read only). The result is printed and saved
# next to this script as verify-result.txt. The URL is typed hidden; it is not printed or saved.
param([Parameter(Mandatory = $true)][long]$EffectiveSlot)
$ErrorActionPreference = 'Stop'
$repo   = 'D:\SaaS\railway-gateway'
$script = Join-Path $PSScriptRoot 'verify_m5.js'
$result = Join-Path $PSScriptRoot 'verify-result.txt'
function Protect-Line([string]$text) {
    # Nothing that identifies the database may reach the screen or the file.
    $text -replace '(postgres(ql)?|rediss?)://\S+', '<URL>' `
          -replace '[A-Za-z0-9-]+(\.[A-Za-z0-9-]+)*\.(rlwy\.net|railway\.app|railway\.internal)(:\d+)?', '<RAILWAY_HOST>' `
          -replace '\b(\d{1,3}\.){3}\d{1,3}(:\d+)?\b', '<IP>' `
          -replace '[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}', '<EMAIL>'
}
$secure = Read-Host 'Paste DATABASE_PUBLIC_URL (production Postgres), then press Enter' -AsSecureString
$url = (New-Object System.Net.NetworkCredential('', $secure)).Password.Trim()
if ($url -notmatch '^postgres(ql)?://[^@\s]+@[^@\s]+$') {
    Write-Host 'STOP: that is not a postgres URL, so nothing was run. Run the command again and paste with a right-click.'
    exit 1
}
Push-Location $repo
try {
    $env:DATABASE_URL = $url
    # The database server itself refuses any write on this connection.
    $env:PGOPTIONS = '-c default_transaction_read_only=on'
    [Console]::OutputEncoding = [System.Text.Encoding]::UTF8
    $ErrorActionPreference = 'Continue'
    $lines = & node $script $EffectiveSlot 2>&1 | ForEach-Object {
        if ($_ -is [System.Management.Automation.ErrorRecord]) { Protect-Line $_.Exception.Message } else { Protect-Line "$_" }
    }
    $code = $LASTEXITCODE
    $lines | ForEach-Object { Write-Host $_ }
    $lines | Out-File -FilePath $result -Encoding utf8
    Write-Host ''
    Write-Host ('DONE. exit code {0} (0 pass, 1 fail, 3 waiting). Saved to {1}' -f $code, $result)
} finally {
    Remove-Item Env:DATABASE_URL, Env:PGOPTIONS -ErrorAction SilentlyContinue
    Remove-Variable url, secure -ErrorAction SilentlyContinue
    Pop-Location
}
